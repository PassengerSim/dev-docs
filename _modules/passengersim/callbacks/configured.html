@dataclass(frozen=True)
class CallbackDefinition:
    """Factory metadata used to validate and construct one callback per simulation."""

    factory: Callable[[BaseModel], Any]
    """Construct a fresh callable from one validated options model."""
    when: Hook | None
    """Lifecycle hook for the constructed callback.

    None for a factory returning a tracer, which then runs on the hooks the
    tracer itself declares (see :func:`~passengersim.callbacks.tracer.tracer_hooks`).
    """
    parameters_model: type[BaseModel]
    """Schema used for runtime validation without changing explicit config."""
    version: str
    """Implementation version checked against an explicit config requirement."""
    output_name: str | None = None
    """Canonical tracer result key, or None for an ordinary callback."""
    



_DEFINITIONS: dict[str, CallbackDefinition] = {}

def register_callback_factory(
    name: str,
    *,
    factory: Callable[[BaseModel], Any],
    when: Hook | None,
    parameters_model: type[BaseModel],
    version: str,
    output_name: str | None = None,
) -> None:
    """Make a factory available to YAML declarations without attaching it.

    Parameters
    ----------
    name : str
        Unique registry name. Duplicate registration is always an error.
    factory : callable
        Receives one validated parameter-model instance and returns a fresh
        callback. Sample hooks receive a simulation; daily hooks also receive
        days prior to departure.
    when : {"initialize_sample", "begin_sample", "daily", "end_sample", "finalize_sample"} or None
        Lifecycle hook at which the returned callback runs. None is allowed
        only for a factory returning a tracer (so `output_name` is required);
        the tracer then runs on every hook it declares.
    parameters_model : type[pydantic.BaseModel]
        Schema for factory options; preferably configured with extra="forbid".
    version : str
        Implementation version checked only when explicitly required in YAML.
    output_name : str or None, optional
        Required canonical result name for a factory returning a GenericTracer.

    Raises
    ------
    ValueError
        If the name is duplicated, the hook is invalid, or the output is reserved.
    TypeError
        If the factory or parameter model is invalid.
    """
    if not isinstance(name, str) or not name or name in _DEFINITIONS:
        raise ValueError(f"Callback registry name {name!r} is empty or already registered")
    if when is None:
        if output_name is None:
            raise ValueError("A callback factory without a hook must return a tracer, so it needs an output_name")
    elif when not in _HOOKS:
        raise ValueError(f"Invalid callback hook: {when!r}")
    if output_name is not None and (not isinstance(output_name, str) or not output_name or output_name in _HOOKS):
        raise ValueError(f"Reserved or empty callback output name: {output_name!r}")
    if not callable(factory) or not isinstance(parameters_model, type) or not issubclass(parameters_model, BaseModel):
        raise TypeError("A callable factory and BaseModel parameter class are required")
    if not isinstance(version, str) or not version:
        raise ValueError("Callback version must be a nonempty string")
    _DEFINITIONS[name] = CallbackDefinition(factory, when, parameters_model, version, output_name)
    



@dataclass(frozen=True)
class _TracerFactory:
    cls: type[GenericTracer]

    def __call__(self, options: BaseModel) -> GenericTracer:
        return self.cls(**options.model_dump())

def options_model_from_signature(cls: type) -> type[BaseModel]:
    """Build a configured-callback options model from a class's constructor.

    Each keyword-capable constructor parameter becomes a field, with the
    parameter's annotation as its type (or ``Any`` if there is none, or it
    cannot be resolved) and its default, if any. A ``name`` parameter is left
    out: a configured tracer always uses its class's result name. Unknown
    options are rejected (``extra="forbid"``).

    Parameters
    ----------
    cls : type
        The class whose ``__init__`` signature is read.

    Returns
    -------
    type[pydantic.BaseModel]
        A model named ``<ClassName>Options``.

    Raises
    ------
    TypeError
        If the constructor takes positional-only parameters, ``*args``, or
        ``**kwargs``, which cannot be described by an options model.
    """
    signature = inspect.signature(cls.__init__)
    try:
        hints = typing.get_type_hints(cls.__init__)
    except Exception:
        # Annotations naming types that are only imported for type checking
        # cannot be resolved; validate those parameters as Any.
        hints = {}
    fields: dict[str, Any] = {}
    for index, (param_name, param) in enumerate(signature.parameters.items()):
        if index == 0 or param_name == "name":
            continue  # self, and the instance-level result name
        if param.kind in (param.POSITIONAL_ONLY, param.VAR_POSITIONAL, param.VAR_KEYWORD):
            raise TypeError(
                f"{cls.__name__}.__init__ has parameter {param_name!r} that cannot be given by keyword; "
                "declare options_model on the class instead"
            )
        annotation = hints.get(param_name, Any)
        default = ... if param.default is param.empty else param.default
        fields[param_name] = (annotation, default)
    return create_model(f"{cls.__name__}Options", __config__=ConfigDict(extra="forbid"), **fields)

def register_tracer_factory(
    cls: type[GenericTracer],
    *,
    parameters_model: type[BaseModel] | None = None,
    version: str | None = None,
    name: str | None = None,
) -> None:
    """Register a tracer using the configuration it declares on its class.

    The output name, options model, and version are read from the class
    attributes ``name``, ``options_model``, and ``options_version``, and the
    hooks from the ``on_<hook>`` methods the class overrides or, for a legacy
    tracer, its ``hook`` attribute (see
    :class:`~passengersim.callbacks.tracer.GenericTracer`). They are defined
    once, beside the tracer's constructor.

    Parameters
    ----------
    cls : type[GenericTracer]
        Tracer class declaring a canonical ``name``, and either hook methods
        or a ``hook``.
    parameters_model : type[pydantic.BaseModel] or None, optional
        Overrides ``cls.options_model``. Fields map to constructor keyword
        arguments. If neither is given, a model is derived from the
        constructor's signature (see :func:`options_model_from_signature`).
    version : str or None, optional
        Overrides ``cls.options_version``, the implementation version checked
        against explicit YAML requirements.
    name : str or None, optional
        Registry name or alias; defaults to ``cls.name``. Does not rename the
        tracer's output.

    Raises
    ------
    TypeError
        If cls is not a GenericTracer subclass, or no options model is
        declared or given and none can be derived from the constructor.
    ValueError
        If registration duplicates a name, the class declares no hook, or it
        declares an invalid hook or output.
    """
    if not isinstance(cls, type) or not issubclass(cls, GenericTracer):
        raise TypeError("Expected a GenericTracer subclass")
    parameters_model = parameters_model or getattr(cls, "options_model", None)
    if parameters_model is None:
        parameters_model = options_model_from_signature(cls)
    hooks = tracer_hooks(cls)
    if not hooks:
        raise ValueError(
            f"{cls.__name__} declares no hook; override one or more on_<hook> methods, "
            "or set the class attribute 'hook'"
        )
    register_callback_factory(
        name or cls.name,
        factory=_TracerFactory(cls),
        # A single-hook tracer keeps its hook on the definition, as before;
        # a tracer with several hooks is attached to all of them at prepare.
        when=hooks[0] if len(hooks) == 1 else None,
        parameters_model=parameters_model,
        version=version or getattr(cls, "options_version", "1"),
        output_name=cls.name,
    )
    



def _resolve(owner: Any) -> list[tuple[str, CallbackDefinition, BaseModel]]:
    """Validate changed explicit declarations without injecting runtime defaults."""
    from passengersim.config.callbacks import CallbackConfig

    raw = {}
    for key, value in owner.config.callbacks.items():
        raw[key] = value.model_dump() if isinstance(value, CallbackConfig) else value
    try:
        signature = json.dumps(list(raw.items()), sort_keys=True, allow_nan=False)
    except (TypeError, ValueError) as error:
        raise ValueError(f"callbacks: parameters must be finite JSON values: {error}") from error
    definitions = getattr(owner, "_configured_definitions", _DEFINITIONS)
    cached = getattr(owner, "_configured_validation", None)
    if cached is not None and cached[0] == signature and cached[1] is definitions:
        return cached[2]
    resolved = []
    for key, value in raw.items():
        name = key
        try:
            declaration = CallbackConfig.model_validate(value)
            if not declaration.enabled:
                continue
            name = declaration.name or key
            if name not in definitions:
                raise ValueError(f"unknown callback registry name {name!r}")
            definition = definitions[name]
            if declaration.version is not None and declaration.version != definition.version:
                raise ValueError(f"version {declaration.version!r} does not match registered {definition.version!r}")
            options = definition.parameters_model.model_validate(copy.deepcopy(declaration.parameters))
        except (ValueError, TypeError) as error:
            raise ValueError(f"callbacks.{key} ({name}): parameters or declaration invalid: {error}") from error
        resolved.append((key, definition, options))
    owner._configured_validation = signature, definitions, resolved
    return resolved


def _definition_hooks(definition: CallbackDefinition) -> tuple[str, ...] | None:
    """The hooks a definition's callback runs on, or None if not known before construction."""
    if definition.when is not None:
        return (definition.when,)
    if isinstance(definition.factory, _TracerFactory):
        return tracer_hooks(definition.factory.cls)
    return None

def preflight(owner: Any) -> dict[str, CallbackDefinition]:
    """Check current declarations and manual tracer collisions without factories.

    This runs before any worker starts, so it also rejects a daily callback
    whose validated options carry a ``priority`` of zero.
    """
    resolved = _resolve(owner)
    outputs = {}
    for key, definition, options in resolved:
        hooks = _definition_hooks(definition)
        # A factory whose hooks are unknown until it runs is checked too: a
        # zero priority is never valid, and prepare would reject it anyway.
        if hooks is None or "daily" in hooks:
            check_daily_priority(getattr(options, "priority", None), f"callbacks.{key}")
        output = definition.output_name
        if output is not None:
            if output in outputs:
                raise ValueError(f"callbacks.{key}: duplicate tracer output {output!r} (also {outputs[output]})")
            outputs[output] = key
    # Only manual tracers: those built by a prior prepare() are about to be replaced.
    for tracer in owner._distinct_tracers(origin="manual"):
        if tracer.name in outputs:
            raise ValueError(
                f"callbacks.{outputs[tracer.name]}: tracer output {tracer.name!r} conflicts with manual tracer"
            )
    # A manual callback registered with name= stores rows under that key too.
    for registration in owner._callback_registry.select(origin="manual"):
        name = getattr(registration.func, "result_name", None)
        if isinstance(name, str) and name in outputs:
            raise ValueError(
                f"callbacks.{outputs[name]}: tracer output {name!r} conflicts with a named manual callback"
            )
    # Copy only definitions referenced by this run, never the global registry.
    declarations = owner.config.callbacks
    return {
        (
            getattr(declarations[key], "name", None)
            or (declarations[key].get("name") if isinstance(declarations[key], dict) else None)
            or key
        ): definition
        for key, definition, _ in resolved
    }

def prepare(owner: Any) -> None:
    """Build fresh callbacks and replace only prior configured attachments.

    Every factory runs, and every result is checked, before anything is
    published. A tracer with several hooks is attached to all of them, and
    its registrations are replaced together on the next prepare.
    """
    from .base import CallbackMixin

    preflight(owner)
    # Callbacks are attached to a scratch object first, which applies the same
    # checks as attaching by hand (hooks, names, priorities, fresh copies).
    temporary = CallbackMixin()
    for key, definition, options in _resolve(owner):
        try:
            callback = definition.factory(options.model_copy(deep=True))
            # A tracer written with hook methods need not be callable itself.
            if not (callable(callback) or isinstance(callback, GenericTracer)):
                raise TypeError("factory must return a callable or a tracer")
            if isinstance(callback, GenericTracer):
                if definition.output_name != callback.name:
                    raise ValueError("declared output_name must match the tracer's canonical name")
            elif definition.output_name is not None:
                raise TypeError("output_name requires a GenericTracer result")
            hook = definition.when
            if hook is None:
                # A tracer's own declared hooks; attaching through the first
                # attaches it to all of them.
                hooks = tracer_hooks(callback)
                if not hooks:
                    raise TypeError("a tracer registered without a hook must declare one")
                hook = hooks[0]
            before = len(temporary._callback_registry.select())
            callback = getattr(temporary, f"{hook}_callback")(callback)
            if not (callable(callback) or isinstance(callback, GenericTracer)):
                raise TypeError("fresh() must return a callable or a tracer")
            if definition.output_name is not None and (
                not isinstance(callback, GenericTracer) or callback.name != definition.output_name
            ):
                raise ValueError("fresh() changed the tracer's canonical output name")
            # Diagnose incompatible hook signatures before any simulation
            # events, for each hook this callback was attached to.
            for registration in temporary._callback_registry.select()[before:]:
                try:
                    signature = inspect.signature(registration.func)
                except (TypeError, ValueError):
                    signature = None  # Some extension callables expose no signature.
                if signature is not None:
                    signature.bind(owner, *([0] if registration.hook == "daily" else []))
        except Exception as error:
            raise ValueError(f"callbacks.{key}: could not construct callback: {error}") from error
    # Publish only after all factories succeed. Manual callbacks and RM callbacks
    # retain their identities and ordering; no stale configured state is reused.
    owner._callback_registry.replace(
        "configured", [(r.hook, r.func, r.owner) for r in temporary._callback_registry.select()]
    )
